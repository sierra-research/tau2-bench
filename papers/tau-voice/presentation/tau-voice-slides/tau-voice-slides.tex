\begin{frame}{Can Voice Agents Actually Do the Job?}

  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \textbf{Full-duplex voice agents} listen and speak simultaneously---the next frontier in conversational AI.

      \vspace{0.8em}
      But today's benchmarks evaluate capabilities \textbf{in isolation}:
      \begin{itemize}
        \item \tautwobench{}: complex grounded tasks, but \textbf{text-only}
        \item Full-Duplex-Bench: turn-taking dynamics, but \textbf{no real tools}
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}
      \textbf{Voice compounds task difficulty:}
      \begin{itemize}
        \item No punctuation, fillers, disfluencies
        \item Accents, background noise, telephony compression
        \item Interruptions, backchannels, real-time turn-taking
      \end{itemize}

      \vspace{0.8em}
      \textcolor{SierraOrange}{\interbold{Missing:}} end-to-end evaluation of voice agents on \textbf{grounded tasks with real-world complexity}.
    \end{column}
  \end{columns}

\end{frame}

\begin{frame}{Headline: a Large Voice--Text Gap Remains}
  \framesubtitle{pass@1 averaged across all domains}

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \vspace{-0.5em}
      \includegraphics[width=\textwidth]{data/tau-voice/results/pass_1_headline_simple.pdf}

      \vspace{0.3em}
      {\small Voice agents retain only \textbf{30--45\%} of text SOTA capability under realistic conditions.}
    \end{column}
    \begin{column}{0.50\textwidth}
      \textbf{Key findings:}
      \begin{itemize}
        \item Best voice (\texttt{grok-voice}, 51\%) nearly matches GPT-4.1 (54\%) under Clean---but \textbf{still 34pp below GPT-5}
        \item Realistic conditions drop performance a further 12pp on average
        \item \texttt{gemini-live-2.5} most \textbf{robust} to degradation (loses only 17\% of Clean)
        \item \texttt{grok-voice} slightly \textbf{highest absolute} scores
        \item All gaps are \textbf{statistically significant} ($p \le 0.002$, paired permutation tests)
      \end{itemize}
    \end{column}
  \end{columns}

\end{frame}

\begin{frame}{Takeaways}

  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{enumerate}
        \item \textbf{The voice--text gap is real and large.} Best voice retains only 45\% of text SOTA---even under clean conditions.

        \vspace{0.3em}
        \item \textbf{It's not just ASR.} Logical errors and hallucinations dominate even with clean audio.

        \vspace{0.3em}
        \item \textbf{Acoustic realism matters unevenly.} Accents are devastating for some models but not others---accessibility concerns.

        \vspace{0.3em}
        \item \textbf{No model does it all.} Fundamental trade-off between responsiveness and restraint.

\begin{frame}{Single-Factor Ablation Results (Retail)}
  \framesubtitle{pass@1 with each factor added on top of Clean}

  \begin{center}
  \begin{small}
  \resizebox{0.92\textwidth}{!}{%
  \begin{tabular}{lccc|c}

    \midrule
    Clean              & 45\% & \textbf{71\%} & 48\% & 55\% \\
    + Noise            & 40\% ($-$4) & \textbf{67\% ($-$4)} & 46\% ($-$2) & 51\% ($-$4) \\
    + Accents          & 44\% ($-$1) & \textbf{60\% ($-$11)} & 30\% ($-$18) & 44\% ($-$10) \\
    + Turn-taking      & 33\% ($-$11) & \textbf{57\% ($-$14)} & 52\% ($+$4) & 47\% ($-$7) \\
    Realistic          & 30\% ($-$15) & \textbf{45\% ($-$26)} & 39\% ($-$10) & 38\% ($-$17) \\
    \bottomrule
  \end{tabular}%
  }
  \end{small}
  \end{center}

  \vspace{0.6em}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{itemize}
        \item \textbf{Accents} largest avg drop ($-$10pp), but \textbf{wildly model-specific}: \texttt{grok-voice} loses 38\% relative; \texttt{gemini-live-2.5} barely moves.
        \item \textbf{Turn-taking} is \texttt{gemini}'s worst factor ($-$11pp, $-$25\% rel.).
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{itemize}
        \item Interactions are non-additive and model-specific (e.g.\ \texttt{grok-voice}'s Realistic drop is \emph{smaller} than the sum of individual factors).
        \item Higher Clean baseline does \emph{not} protect against degradation.
      \end{itemize}
    \end{column}
  \end{columns}

\begin{frame}{Failure Mode Details}
  \framesubtitle{Common qualitative patterns}

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Authentication bottleneck:}
      \begin{itemize}
        \item Fail to transcribe names/emails even when spelled letter-by-letter
        \item Blocks downstream actions; compounds with accents
      \end{itemize}

      \vspace{0.2em}
      \textbf{Hallucinated completions:}
      \begin{itemize}
        \item ``I've updated your address''---no tool call made
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Multi-step tracking failures:}
      \begin{itemize}
        \item Completes part of a request, forgets the rest
      \end{itemize}

      \vspace{0.2em}
      \textbf{Compounding under realistic conditions:}
      \begin{itemize}
        \item Unresponsive after repeated auth failures
        \item Verbosity causes timeouts on complex tasks
      \end{itemize}
    \end{column}
  \end{columns}

  \vspace{0.4em}
  {\footnotesize $\sim$95\% of agent-attributable failures reflect domain-agnostic conversational skills (spelling, grounding, honesty, tracking, arithmetic)---not domain policy gaps.}

\end{frame}

%% ---------------------------------------------------------------------------

\begin{frame}{Limitations}

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Language \& speech:}
      \begin{itemize}
        \item English-only, TTS-synthesized voices
        \item Accent findings via TTS personas are \textit{indicative}, not definitive
      \end{itemize}

      \vspace{0.6em}
      \textbf{Evaluation scope:}
      \begin{itemize}
        \item Task completion + conversational dynamics
        \item Not measured: agent speech naturalness, user satisfaction, partial task success
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Simulator fidelity:}
      \begin{itemize}
        \item More patient than real users, perfect memory, instant tool calls
        \item Decoupled from wall-clock $\rightarrow$ p95 sim latency $\sim$1.5s
        \item Human raters scored sim \textbf{3.1/4} on naturalness (60 sims; 83\% $\ge$3)
      \end{itemize}

      \vspace{0.6em}
      \textbf{Transcript injection:}
      \begin{itemize}
        \item ASR bypassed on agent side $\rightarrow$ default mode is an \textbf{upper bound} on real-world performance
        \item Optional ASR-enabled mode confirms the gaps persist
      \end{itemize}
    \end{column}
  \end{columns}

\end{frame}
